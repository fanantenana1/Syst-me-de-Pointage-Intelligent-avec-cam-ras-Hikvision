def normalize_employee_id(emp_id):
    """Normalise l'ID employé pour la comparaison (enlève zéros, espaces, etc.)"""
    if not emp_id or emp_id == "-":
        return None
    # Convertir en string et enlever espaces
    emp_id = str(emp_id).strip()
    # Retourner version originale ET version sans zéros de tête
    return emp_id, emp_id.lstrip('0')

def employee_ids_match(id1, id2):
    """Compare deux IDs employés de manière flexible"""
    norm1 = normalize_employee_id(id1)
    norm2 = normalize_employee_id(id2)
    
    if not norm1 or not norm2:
        return False
    
    # Comparer les deux versions (avec et sans zéros)
    return (norm1[0] == norm2[0] or norm1[0] == norm2[1] or 
            norm1[1] == norm2[0] or norm1[1] == norm2[1])

def get_presence_info(major, minor):
    """Retourne (méthode, validé) ou None"""
    return PRESENCE_EVENTS.get((major, minor))

def get_period_of_day(hour):
    """Retourne 'Jour' ou 'Soir' selon l'heure"""
    if 6 <= hour < 18:
        return "Jour"
    else:
        return "Soir"

  // ESP32 Stream
  function startEsp32Stream() {
    stopEsp32Stream();
    if (ipWebcamURL) {
      esp32StreamInterval = setInterval(() => {
        esp32Preview.src = ipWebcamURL + "?rand=" + Math.random();
      }, 500);
    }
  }

  function stopEsp32Stream() {
    if (esp32StreamInterval) {
      clearInterval(esp32StreamInterval);
      esp32StreamInterval = null;
    }
  }

  // Capture from camera
  function captureFromCamera(cameraType) {
    let sourceImg;
    
    if (cameraType === 'esp32') {
      sourceImg = esp32Preview;
    } else if (cameraType === 'hikvision') {
      sourceImg = hikvisionPreview;
    }

    if (!sourceImg) {
      alert('Erreur: Source caméra non trouvée');
      return;
    }

    const canvas = document.createElement("canvas");
    canvas.width = sourceImg.naturalWidth || 640;
    canvas.height = sourceImg.naturalHeight || 480;
    const ctx = canvas.getContext("2d");
    
    try {
      ctx.drawImage(sourceImg, 0, 0);

      canvas.toBlob((blob) => {
        if (!blob) {
          alert('Erreur lors de la capture de l\'image');
          return;
        }

        const file = new File([blob], `capture_${cameraType}_${Date.now()}.png`, { type: 'image/png' });
        const dataTransfer = new DataTransfer();
        
        // Add existing files
        for (let i = 0; i < photosInput.files.length; i++) {
          dataTransfer.items.add(photosInput.files[i]);
        }
        
        // Add new capture
        dataTransfer.items.add(file);
        photosInput.files = dataTransfer.files;

        // Add to preview
        const reader = new FileReader();
        reader.onload = (e) => {
          const previewImg = document.createElement('img');
          previewImg.src = e.target.result;
          previewContainer.appendChild(previewImg);
        };
        reader.readAsDataURL(file);

        alert(`✅ Photo capturée depuis ${cameraType === 'esp32' ? 'ESP32' : 'Hikvision'}!`);
      }, 'image/png');
    } catch (error) {
      console.error('Capture error:', error);
      alert('❌ Erreur lors de la capture: ' + error.message);
    }
  }

  // Preview multiple images from file input
  function previewMultipleImages(event) {
    previewContainer.innerHTML = "";
    const files = event.target.files;
    for (let i = 0; i < files.length; i++) {
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = document.createElement('img');
        img.src = e.target.result;
        previewContainer.appendChild(img);
      };
      reader.readAsDataURL(files[i]);
    }
  }
